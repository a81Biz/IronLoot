    build:
      context: ./src/nginx
      dockerfile: Dockerfile
    container_name: ironloot-nginx-tls

    ports:
      - "${NGINX_HTTP_PORT:-80}:80"
      - "${NGINX_HTTPS_PORT:-443}:443"

    depends_on:
      api:
        condition: service_healthy
      admin:
        condition: service_started
      base:
        condition: service_started
      client:
        condition: service_started
    networks:
      - ironloot-network
